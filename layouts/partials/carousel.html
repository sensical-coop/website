{{- /* Reusable image carousel: slides on a scroll-snap track, so swiping works without JS; scripts.js adds the arrows, dots, keyboard and the eased slide animation. Context: dict "page" page whose bundle resources are looked up first, then assets/ (optional) "images" list of paths/URLs, or maps with "src", "alt", "caption" "ratio" slide aspect ratio, as CSS (default "16/10") "fit" "cover" (default, crops) or "contain" (whole image, for drawings) "label" accessible name (default: T "Image gallery") */ -}} {{- $page := .page -}} {{- $fit := .fit | default "cover" -}} {{- $ratio := .ratio | default "16/10" -}} {{- $slides := slice -}} {{- range .images -}} {{- $s := . -}} {{- if not (reflect.IsMap $s) }}{{ $s = dict "src" $s }}{{ end -}} {{- $src := $s.src | default "" | string -}} {{- $url := $src | relURL -}} {{- $srcset := "" -}} {{- with partial "image-resource.html" (dict "page" $page "src" $src) -}} {{- $url = .RelPermalink -}} {{- if and (eq .ResourceType "image") (in (slice "jpeg" "png" "webp" "tiff" "bmp") .MediaType.SubType) -}} {{- $set := slice -}} {{- $img := . -}} {{- range slice 800 1600 2400 -}} {{- if lt . $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q80" .) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink .) -}} {{- end -}} {{- end -}} {{- $full := $img.Resize (printf "%dx webp q80" (math.Min $img.Width 2400 | int)) -}} {{- $url = $full.RelPermalink -}} {{- $set = $set | append (printf "%s %dw" $full.RelPermalink $full.Width) -}} {{- $srcset = delimit (uniq $set) ", " -}} {{- end -}} {{- end -}} {{- if $src }}{{ $slides = $slides | append (dict "url" $url "srcset" $srcset "alt" ($s.alt | default "") "caption" ($s.caption | default "")) }}{{ end -}} {{- end -}} {{- with $slides }} {{- $total := len . }}
{{- range $i, $s := . }} {{- end }}
{{- if gt $total 1 }}
01 / {{ printf "%02d" $total }}
{{- end }}
{{- end -}}
